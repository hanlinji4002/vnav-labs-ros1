cmake_minimum_required(VERSION 3.0.2)
project(controller_prehover)
add_compile_options(-std=c++11 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs geometry_msgs trajectory_msgs mav_msgs)
find_package(Eigen3 REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp nav_msgs geometry_msgs trajectory_msgs mav_msgs)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(controller_prehover_node src/controller_prehover_node.cpp)
add_dependencies(controller_prehover_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(controller_prehover_node ${catkin_LIBRARIES})
